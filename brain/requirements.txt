# Raspberry Pi 5 hardware-control dependencies for the LocalAiRobot brain.
# Install into a virtualenv (Raspberry Pi OS Bookworm is externally-managed):
#   python3 -m venv .venv && .venv/bin/pip install -r requirements.txt
#
# The AI stack (Ollama, whisper.cpp, Piper, arecord/aplay) is NOT pip-installed --
# it is set up by scripts/install_stack.sh and invoked as external processes.
#
# picamera2 is also NOT pip-installed: it comes from apt (python3-picamera2) and is visible to
# this venv because install_stack.sh creates the venv with --system-site-packages.

adafruit-circuitpython-servokit   # ESC + steering servo via PCA9685 (pulls in Blinka, PCA9685, motor)
gpiozero                          # 3x HC-SR04P ultrasonic sensors (front / left / right)
lgpio                             # gpiozero pin-factory backend for the Raspberry Pi 5
Pillow                            # camera frame letterbox/JPEG + procedural face rendering
numpy                             # RGB565 packing for the DSI/HDMI framebuffer face fallback
pyserial                          # USB-C face panel (turing-smart-screen serial protocol)
